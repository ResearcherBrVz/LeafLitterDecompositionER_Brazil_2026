###### PROLONGED DRY HALTS LEAF LITTER DECOMPOSITION IN AN EPHEMERAL RIVER OF THE BRAZILIAN TROPICAL SEMIARID REGION###

## Linear Mixed-Effects (lme) Models 
 
#Load packages

##Load Data
LLD_data = read.csv("data_LLD_Brasil.csv", sep=";")
str(LLD_data)
 
##Explore data
LLD_data$Site = as.factor(LLD_data$Site)
unique(LLD_data$Site) # 15 sites
unique(LLD_data$Period) # 3 campaigns 
##Call periods
LLD_data$Period <- factor(LLD_data$Period, levels=c('Dry', 'Very Dry', 'Wet'))

# All sampling sites remained in a dry riverbed state (no surface flow).
# We thus defined three hydroclimatic phases representing contrasting regional climatic conditions
# These phases represent differences in external "climatic forcing"
 
# Structure of data:
#  ?? Period (3) : climatic variables
#   ?? Site (15) : spatial variables
# 	?? Species (4)
#  	?? 3 replicates ? k : sediment variables
 
# Split by habitat
data_DS = subset(LLD_data, Habitat=="DS")
data_IP = subset(LLD_data, Habitat=="IP")
 
 
### Mixed effect models ---> OK FOR DATA ##############
 
# 1. DS (dry riverbeds)####
DS_m1 <- lmer(
  log(k) ~ Species * Period +
   	(1 | Site),  # account for spatial autocorrelation of data
  data = data_DS)
 
anova(DS_m1)
summary(DS_m1)
 
# as we used log(k), we use exp() to calculate the effects in %
emm_DS <- emmeans(DS_m1, ~ Species | Period)
emm_DS <- as.data.frame(emm_DS)
emm_DS$k <- exp(emm_DS$emmean)
 
# 2. IP (isolated pools) ####
colnames(data_IP)
 
IP_m1 <- lmer(
log(k) ~ Species * Period +
 (1 | Site),  # account for spatial autocorrelation of data
data = data_IP[!data_IP$Period=="Very Dry",]) # excluding very dry period
 
anova(IP_m1)
 
emm_IP <- emmeans(IP_m1, ~ Species | Period)
emm_IP <- as.data.frame(emm_IP)
emm_IP$k <- exp(emm_IP$emmean)
 
#plots (Figure 4)
# Summary for plots ####
 
data_IP_summary <- data_IP %>%
  group_by(Species, Period) %>%
  summarise(
	mean_k = mean(k, na.rm = TRUE),
	se = sd(k, na.rm = TRUE) / sqrt(n()),
	n = n())
 
data_DS_summary <- data_DS %>%
  group_by(Species, Period) %>%
  summarise(
	mean_k = mean(k, na.rm = TRUE),
	se = sd(k, na.rm = TRUE) / sqrt(n()),
	n = n())
 
# With log (k) ---------
 
plot_DS = ggplot(data_DS_summary, aes(x = Period, y = log(mean_k), colour = Species)) +
 
  geom_errorbar(aes(ymin = log(mean_k - se), ymax = log(mean_k + se),
    	            width = 0.05), linewidth = 0.8 ) + geom_point(size=0.5) +
  geom_line(aes(group = Species), linewidth = 0.7) +
  ggtitle("A. Dry riverbeds") + ylim(-10, -2.5)+
  labs(
	x = "Experiment",
	y = expression(log (k)))+ theme_cowplot() +
  theme(
	panel.grid = element_blank(),
    panel.background = element_blank(),
    plot.background = element_blank(),
	plot.title = element_text(size = 16, face = "plain"),
	axis.text = element_text(size = 11),
	axis.title = element_text(size = 12),
	legend.text = element_text(size = 10, face = "italic"),
	legend.title = element_text(size = 12)
  )
 
 data_IP_summary <- data_IP_summary %>%
  filter(Period != "Very Dry")

plot_IP = ggplot(data_IP_summary, aes(x = Period, y = log(mean_k), colour = Species)) +
 
  geom_errorbar(aes(ymin = log(mean_k - se), ymax = log(mean_k + se),
                    width = 0.05), linewidth = 0.8 ) + geom_point(size=1) +
  geom_line(aes(group = Species), linewidth = 0.7) +
  ggtitle("B. Isolated pools") + ylim(-10, -2.5)+
  labs(
	x = "Experiment",
	y = "") + theme_cowplot() +
  theme(
	panel.grid = element_blank(),
    panel.background = element_blank(),
    plot.background = element_blank(),
	plot.title = element_text(size = 16, face = "plain"),
	axis.text = element_text(size = 11),
	axis.title = element_text(size = 12),
	legend.text = element_text(size = 10, face = "italic"),
	legend.title = element_text(size = 12)
  )
 
ggarrange(plot_DS, plot_IP, ncol=2, common.legend=T, legend ="right") 
 
ggsave(
  "int_k.jpg",

  width = 20,
  height = 10,
  units = "cm",
  dpi = 300
)
