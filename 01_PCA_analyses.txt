# PROLONGED DRY HALTS LEAF LITTER DECOMPOSITION IN AN EPHEMERAL RIVER OF THE BRAZILIAN TROPICAL SEMIARID REGION###

# Install binary versions with Windows compatibility 
install.packages(
  c("rlang", "vctrs", "dplyr", "tidyr", "ggpubr", "factoextra"),
  type = "binary"
)
 
# PCA analyses with physicochemical variables. We performed one analyse per habitat
 
#Load packages
library(ggplot2)
library(dplyr)
library(ggpubr)
library(factoextra)
library(FactoMineR)
library(patchwork)
library(car)


 
# read and pre-process in data
ml = read.csv("PCA_data.csv", header = T, sep = ";") %>%
  mutate(Species = as.factor(Species),
     	Habitat = as.factor(Habitat),
     	Period = as.factor(Period),
     	Site = as.factor(Site))
str(ml)
summary(ml)
ml
ml <- ml %>%
  dplyr::mutate(
	Period = dplyr::recode(
  	Period,
      "LDP" = "WET",
      "EDP" = "DRY",
      "MDP" = "VERY DRY"
	)
  )
 
# Subset Dry riverbeds (DS)
dry0 <- subset(ml, Habitat == "DS")
dry0
dry_pca_df <- na.omit(dry0[, c(4,5, 6, 9:12)])
# Order levels of periods
dry_pca_df$Period <- factor(
  dry_pca_df$Period,
  levels = c("DRY", "VERY DRY", "WET")
)
#Eliminate period columns for create numeric dataframe
datos_pca <- dry_pca_df[, -1]  
 
#check collinearity with VIF function 
vif(
  lm(D_P ~ moisture + sed_temp + sed_pH +
   	Rainfall + max_temp,
 	data = datos_pca)
)
 
# Run PCA used escalated units 
drypca <- PCA(
  datos_pca,
  scale.unit = TRUE,
  graph = FALSE
)
summary(drypca)

# Explore contributions
fviz_contrib(drypca, choice = "var", axes = 1)
fviz_contrib(drypca, choice = "var", axes = 2)
summary(drypca)
 
# Subset for Isolated pools
pool0 <- subset(ml, Habitat == "IP")
pool0<- subset(pool0, Period != "VERY DRY") ##filter mdp because the period had only 1 pool
pool0
 
pool_pca_df <- na.omit(pool0[, c(4, 6, 7, 11, 13:16)])
pool_pca_df

# Order levels of periods
pool_pca_df$Period <- factor(
  pool_pca_df$Period,
  levels = c("DRY", "WET")
)
 
#Eliminate period columns
datos_pca_p <- pool_pca_df[, -1]
datos_pca_p
 
#check collinearity
 
vif(
  lm( Rainfall ~ temp_IP + pH_IP +
   	do + con + volu + max_temp,
 	data = datos_pca_p)
)
 
##PCA
poolpca <- PCA(
  datos_pca_p,
  scale.unit = TRUE,
  graph = FALSE
)
 
# Explore contributions 
fviz_contrib(poolpca, choice = "var", axes = 1)
fviz_contrib(poolpca, choice = "var", axes = 2)
 
 #Plots (Figure 3)
#Colors for periods
period_colors <- c(
  "DRY"  	= "#F4A6B8",  
  "VERY DRY" = "#FF8C00",  
  "WET"  	= "#56B4E9"  
)
 
# Shapes for period
period_shapes <- c(
  "DRY" = 16,    	
  "VERY DRY" = 17,  
  "WET" = 15     	
)
 
# Biplot PCA(Dim 1 vs Dim 2)
poolpca12 <- fviz_pca_biplot(
  poolpca,
  habillage = pool_pca_df$Period,
  addEllipses = FALSE,
  label = "var",
  mean.point = FALSE,
  title = " ",
  col.var = "black",
  labelsize = 4
)   + labs(
  x = "PCA 1 (37.7 %)",
  y = "PCA 2 (21.9)"
) +
  scale_color_manual(values = period_colors) +
  scale_shape_manual(values = period_shapes)
poolpca12
 ggsave("PCA_pool_2026a.jpeg", width = 13, height = 9, units = "cm", bg="white")

drypca12 <- fviz_pca_biplot(
  drypca,
  habillage = dry_pca_df$Period,
  addEllipses = FALSE,
  label = "var",
  mean.point = FALSE,
  title = " ",
  col.var = "black",
  labelsize = 4
) +  labs(
  x = "PCA 1 (31.6 %)",
  y = "PCA 2 (26.2)"
) +
  scale_color_manual(values = period_colors) +
  scale_shape_manual(values = period_shapes)
drypca12
 (drypca12 | poolpca12) +
  plot_layout(guides = "collect") &
  theme(legend.position = "bottom")
 
ggsave("PCA_drysediments_2026a.jpeg", width = 13, height = 9, units = "cm", bg="white")
