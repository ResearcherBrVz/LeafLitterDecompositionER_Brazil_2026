#Explore differences of anatomical traits (%undegraded midrib area and total phenolic compounds) between species (analyse performed in undegraded leaves)
#load data %UM (undegraded midrid %)
ACLc <- (read.csv("UM.csv", sep = ";"))
ACLc

# Run Anova
attach(ACLc)
summary(ACLc)
ACLc$Especie<-as.factor(ACLc$Especie)
str(ACLc)
 
mACL0 <- aov(log(ACL.) ~ Especie, data = ACLc)
summary(mACL0)

#Check residuals normality 
res_acl <- residuals(mACL0)
shapiro.test(res_acl)
summary(mACL0)
#run post hoc test 
tuk <- TukeyHSD(mACL0)
tuk

#Explore  groups formed for Tukey test
letters <- multcompLetters4(mACL0, tuk)
letters
 
##TPC ( total phenolic compounds)

)

tpc

tpc
 tpc$ESPECIE<-as.factor(tpc$ESPECIE)

mtpc0 <- aov(tpc$C_FENOLIC ~ tpc$ESPECIE, data = tpc)
summary(mtpc0)

#Check residual normality

shapiro.test(res_tpc)
tuk <- TukeyHSD(mtpc0)
tuk

letters
